{========================================}
{  TD序列(汤姆·迪马克) 改进版 主图公式     }
{  优化点: 个股趋势过滤 + VSA量价确认      }
{         + 真COUNTDOWN(TD13) + ATR止损    }
{  颜色: 红=买/涨  绿=卖/跌 (A股习惯)      }
{  信号无未来函数, 建议用前复权数据         }
{========================================}

{--- 可调参数(想调直接改数字) ---}
N:=4;     {SETUP比较周期, 经典TD=4}
M:=9;     {SETUP完成计数}
P:=13;    {COUNTDOWN完成计数}

{==========================================}
{ 1. SETUP计数 (与原公式一致, 保留NT==9信号)}
{==========================================}
UP:=C>REF(C,N);            {上涨条件: 收盘>4根前收盘}
DN:=C<REF(C,N);            {下跌条件: 收盘<4根前收盘}
NTU:=BARSLASTCOUNT(UP);    {上涨连续计数}
NTD:=BARSLASTCOUNT(DN);    {下跌连续计数}
SELL_SETUP:=NTU=M;         {卖SETUP完成(上涨9根)}
BUY_SETUP :=NTD=M;         {买SETUP完成(下跌9根)}

{==========================================}
{ 2. 个股趋势过滤 (替代指数状态, 实证更有效)}
{==========================================}
MA60:=MA(C,60);
UP_TREND:=C>MA60;                     {多头}
DN_TREND:=C<MA60;                     {空头}
SELL_OK:=SELL_SETUP AND UP_TREND;     {多头末端才给卖点}
BUY_OK :=BUY_SETUP  AND DN_TREND;     {空头末端才给买点}

{==========================================}
{ 3. VSA量价确认 (契合威科夫/量价分析)      }
{==========================================}
VMA20:=MA(V,20);
SELL_VOL:=V<REF(V,1) OR V<VMA20;      {滞涨: 第9根不继续放量}
BUY_VOL :=V<VMA20;                    {止跌: 第9根缩量}
SELL_SIG:=SELL_OK AND SELL_VOL;       {最终卖出信号}
BUY_SIG :=BUY_OK  AND BUY_VOL;        {最终买入信号}

{==========================================}
{ 4. 真COUNTDOWN (TD13, 替换原公式伪"13")   }
{    卖CD: SETUP后累计 CLOSE<=REF(C,N)的棒 }
{    买CD: SETUP后累计 CLOSE>=REF(C,N)的棒 }
{    出现反向SETUP自动取消(重置)            }
{    注: "创新高/新低重置"为可选, 见文末说明}
{==========================================}
DBS:=BARSLAST(SELL_SETUP);                        {距上次卖SETUP}
IN_SELL_CD:=BARSLAST(SELL_SETUP)<BARSLAST(BUY_SETUP); {处于卖CD阶段}
SELL_CNT:=SUM(C<=REF(C,N), DBS);                  {卖SETUP后累计合格棒}
TD13_SELL:=IN_SELL_CD AND SELL_CNT=P AND DBS>=P;  {卖COUNTDOWN满13}

DBB:=BARSLAST(BUY_SETUP);                         {距上次买SETUP}
IN_BUY_CD:=BARSLAST(BUY_SETUP)<BARSLAST(SELL_SETUP);{处于买CD阶段}
BUY_CNT:=SUM(C>=REF(C,N), DBB);                   {买SETUP后累计合格棒}
TD13_BUY:=IN_BUY_CD AND BUY_CNT=P AND DBB>=P;     {买COUNTDOWN满13}

{==========================================}
{ 5. ATR止损止盈标记                        }
{==========================================}
TRR:=MAX(MAX(H-L,ABS(H-REF(C,1))),ABS(L-REF(C,1)));
ATR:=MA(TRR,14);
SELL_STOP:=H+2*ATR;        {卖信号后上破此价止损}
BUY_STOP :=L-2*ATR;        {买信号后下破此价止损}

{==========================================}
{ 6. 绘制                                   }
{==========================================}
{SETUP进行中计数 1-8 (灰, 实时显示进度)}
DRAWNUMBER(UP_TREND AND BETWEEN(NTU,1,8), H*1.008, NTU),COLORGRAY;
DRAWNUMBER(DN_TREND AND BETWEEN(NTD,1,8), L*0.992, NTD),COLORGRAY;
{SETUP完成 第9根 (卖=绿 / 买=红)}
DRAWNUMBER(UP_TREND AND NTU=M, H*1.012, M),COLORGREEN;
DRAWNUMBER(DN_TREND AND NTD=M, L*0.988, M),COLORRED;
{最终信号文字 (经趋势+量价双重过滤)}
DRAWTEXT(SELL_SIG, H*1.03, '卖出'),COLORGREEN;
DRAWTEXT(BUY_SIG , L*0.97, '买入'),COLORRED;
{真COUNTDOWN完成 TD13}
DRAWTEXT(TD13_SELL, H*1.04, 'TD13'),COLORGREEN;
DRAWTEXT(TD13_BUY , L*0.96, 'TD13'),COLORRED;
{止损价位数字}
DRAWNUMBER(SELL_SIG, SELL_STOP, SELL_STOP),COLORGREEN;
DRAWNUMBER(BUY_SIG , BUY_STOP , BUY_STOP ),COLORRED;

{==========================================}
{ 使用说明:                                }
{ 1. 通达信公式管理器→新建→主图叠加→粘贴    }
{ 2. 建议前复权数据使用                     }
{ 3. 颜色统一: 绿=卖/跌, 红=买/涨           }
{ 4. 信号=经"趋势+量价"过滤后的 卖出/买入    }
{    TD13=真COUNTDOWN完成(更强的反转确认)    }
{ 5. 可选创新高重置(学术严谨版):            }
{    在TD13_SELL条件加 AND HHV(H,DBS)<REF(HHV(H,9),DBS)}
{    (即SETUP后未创SETUP区间新高)           }
{==========================================};
